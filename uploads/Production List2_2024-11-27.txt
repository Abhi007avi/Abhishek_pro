using System;
using System.IO;
using System.Net.Http;
using System.Threading.Tasks;
using CsvHelper;
using CsvHelper.Configuration;
using MySql.Data.MySqlClient;
using System.Collections.Generic;
using System.Dynamic;
using System.Globalization;
using System.Net;

class Program
{
    static async Task Main(string[] args)
    {
        string connectionString = "Server=192.168.0.90;Database=rimm;User ID=user;Password=user;Port=3306;";
        HttpClientHandler handler = new HttpClientHandler
        {
            CookieContainer = new CookieContainer()
        };

        using (HttpClient client = new HttpClient(handler))
        {
            client.Timeout = TimeSpan.FromMinutes(5);
            string requestUri = "http://192.168.0.41/DataLogs?Path=logging.csv&Action=DOWNLOAD";
            client.DefaultRequestHeaders.Referrer = new Uri("http://192.168.0.41/Portal/Portal.mwsl?PriNav=DataLogs");

            try
            {
                handler.CookieContainer.Add(new Uri(requestUri), new Cookie("sessionid", "your-session-id"));
                HttpResponseMessage response = await client.GetAsync(requestUri);
                response.EnsureSuccessStatusCode();

                byte[] fileBytes = await response.Content.ReadAsByteArrayAsync();
                string filePath = "D:\\logging.csv";
                await File.WriteAllBytesAsync(filePath, fileBytes);

                // Parse the CSV and insert into database directly
                await InsertCsvDataDirectly(filePath, connectionString);
            }
            catch (HttpRequestException e)
            {
                Console.WriteLine("Request error: " + e.Message);
            }
            catch (IOException e)
            {
                Console.WriteLine("IO error: " + e.Message);
            }
            catch (TaskCanceledException e)
            {
                Console.WriteLine("Request timed out: " + e.Message);
            }
        }
    }

    static async Task InsertCsvDataDirectly(string csvFilePath, string connectionString)
    {
        using (var reader = new StreamReader(csvFilePath))
        using (var csv = new CsvReader(reader, new CsvConfiguration(CultureInfo.InvariantCulture)
        {
            HasHeaderRecord = false // Assuming no header record
        }))
        {
            var records = csv.GetRecords<dynamic>();

            foreach (var record in records)
            {
                var expandoRecord = (IDictionary<string, object>)record;

                // Safely access columns by index, assuming columns are present
                //string col1 = expandoRecord.ContainsKey("0") ? Convert.ToString(expandoRecord["0"]) : string.Empty;
                //string col2 = expandoRecord.ContainsKey("1") ? Convert.ToString(expandoRecord["1"]) : string.Empty;
                string dateColumn = expandoRecord.ContainsKey("1") ? Convert.ToString(expandoRecord["1"]) : string.Empty;

                // Convert date to desired format (YYYY-MM-DD)
                string formattedDate =  ConvertDateFormat(dateColumn);

                // Concatenate columns 4 to 14 (0s and 1s) for pShot
                string concatenatedBits = "";
                for (int i = 3; i <= 13; i++) // Access columns 4 to 14
                {
                    if (expandoRecord.ContainsKey(i.ToString()))
                    {
                        concatenatedBits += Convert.ToString(expandoRecord[i.ToString()]);
                    }
                }

                // Insert data directly into the table
                using (var connection = new MySqlConnection(connectionString))
                {
                    await connection.OpenAsync();

                    string sql = @"INSERT INTO production_shot_instance_master_buffering  (pShot, pPortNo, pBufferingDate)
                                   VALUES (@pShot, @pPortNo, @pBufferingDate)";

                    using (var command = new MySqlCommand(sql, connection))
                    {
                        command.Parameters.AddWithValue("@pShot", concatenatedBits);
                        command.Parameters.AddWithValue("@pPortNo", "192.168.0.90"); // Assuming a constant value for pPortNo
                        command.Parameters.AddWithValue("@pBufferingDate", formattedDate);

                        await command.ExecuteNonQueryAsync();
                    }
                }
            }
        }
    }

    static string ConvertDateFormat(string dateStr)
    {
        if (DateTime.TryParseExact(dateStr, "MM-dd-yyyy", CultureInfo.InvariantCulture, DateTimeStyles.None, out DateTime parsedDate))
        {
            return parsedDate.ToString("yyyy-MM-dd");
        }
        else
        {
            // Log or handle the date parsing error
            Console.WriteLine($"Unable to parse date string '{dateStr}'");
            return parsedDate.ToString("yyyy-MM-dd");
        }
    }
}
